$("pasteReplaceBtn").onclick = async () => { const editor = $("codeEditor"); editor.focus(); try { const text = await navigator.clipboard.readText(); if (text) { editor.value = text; setMjState("hack", "Clipboard pasted successfully!"); } } catch (err) { // Fallback for strict mobile browser security policies const manual = prompt("Paste your code here:"); if (manual !== null) { editor.value = manual; setMjState("hack", "Code loaded via fallback!"); } } };
